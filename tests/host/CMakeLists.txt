cmake_minimum_required(VERSION 3.22)
project(BSMLImageLifetimeTests LANGUAGES CXX)
enable_testing()

add_executable(image_lifetime_tests
    image-lifetime.cpp
    ../../src/BSML/Animations/AnimationStateUpdater.cpp
)
target_compile_features(image_lifetime_tests PRIVATE cxx_std_20)
target_include_directories(image_lifetime_tests PRIVATE stubs ../../src/Helpers)
# Assertions are the test oracle in every build configuration.
if(MSVC)
    target_compile_options(image_lifetime_tests PRIVATE /UNDEBUG)
else()
    target_compile_options(image_lifetime_tests PRIVATE -UNDEBUG)
endif()
add_test(NAME image_lifetime COMMAND image_lifetime_tests)

add_executable(gif_validation_tests gif-validation.cpp)
target_compile_features(gif_validation_tests PRIVATE cxx_std_20)
target_include_directories(gif_validation_tests PRIVATE ../../src/BSML/Animations/GIF)
if(MSVC)
    target_compile_options(gif_validation_tests PRIVATE /UNDEBUG)
else()
    target_compile_options(gif_validation_tests PRIVATE -UNDEBUG)
endif()
add_test(NAME gif_validation COMMAND gif_validation_tests "${CMAKE_CURRENT_SOURCE_DIR}/../quest/fixtures")

add_executable(active_image_iteration_tests active-image-iteration.cpp)
target_compile_features(active_image_iteration_tests PRIVATE cxx_std_20)
target_include_directories(active_image_iteration_tests PRIVATE stubs ../../src/BSML/Animations)
if(MSVC)
    target_compile_options(active_image_iteration_tests PRIVATE /UNDEBUG)
else()
    target_compile_options(active_image_iteration_tests PRIVATE -UNDEBUG)
endif()
add_test(NAME active_image_iteration COMMAND active_image_iteration_tests)

find_package(Threads REQUIRED)
add_executable(gif_stream_tests gif-stream.cpp ../../src/BSML/Animations/AnimationInfo.cpp)
target_compile_features(gif_stream_tests PRIVATE cxx_std_20)
target_include_directories(gif_stream_tests PRIVATE stubs ../../shared ../../src/BSML/Animations/GIF)
target_link_libraries(gif_stream_tests PRIVATE Threads::Threads)
if(MSVC)
    target_compile_options(gif_stream_tests PRIVATE /UNDEBUG)
else()
    target_compile_options(gif_stream_tests PRIVATE -UNDEBUG)
endif()
add_test(NAME gif_stream COMMAND gif_stream_tests)
set_tests_properties(gif_stream PROPERTIES TIMEOUT 15)

add_executable(indexed_frame_tests indexed-frame.cpp)
target_compile_features(indexed_frame_tests PRIVATE cxx_std_20)
target_include_directories(indexed_frame_tests PRIVATE ../../src/BSML/Animations/GIF)
if(MSVC)
    target_compile_options(indexed_frame_tests PRIVATE /UNDEBUG)
else()
    target_compile_options(indexed_frame_tests PRIVATE -UNDEBUG)
endif()
add_test(NAME indexed_frame COMMAND indexed_frame_tests)
